8enum class IpcMessageType : uint32_t
9{
10 Handshake = 0x01,
11 MidiData = 0x02,
12 AudioConfig = 0x03,
13 Shutdown = 0x04,
14 Heartbeat = 0x05, // Bridge → Core, keepalive (prevents read timeout)
15 WindowPos = 0x06, // bidirectional: Bridge→Core (reports bounds), Core→Bridge (sets bounds)
16 RequestState = 0x07, // Core → Bridge: request plugin state dump
17 StateData = 0x08, // Bridge → Core: plugin state binary payload
18 SetState = 0x09, // Core → Bridge: restore plugin state from binary payload
19 SetWindowTitle = 0x0A, // Core → Bridge: update the Bridge window title string
20};

22// Shared memory layout: header + audio I/O buffers (stereo, up to 4096 samples)
23// Core writes audioIn + signals Bridge; Bridge processes + writes audioOut + signals Core.
24struct SharedAudioLayout
