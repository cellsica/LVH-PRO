5// =====================================================================
6// LanguageManager — centralised UI string localisation (i18n)
7//
8// Usage:
9// LvhStr ("STR_LOAD") → "LOAD" (en) / "ロード" (ja)
10//
11// Initialise once at startup (before any windows open):
12// LanguageManager::getInstance().init (appProperties_.getUserSettings());
13//
14// Change language at runtime:
15// LanguageManager::getInstance().setLanguage (lang, prefs);
16//
17// Adding a new string:
18// 1. Add a { "STR_MY_KEY", "English text", "日本語テキスト" } entry
19// to the table inside get().
20// 2. Call LvhStr ("STR_MY_KEY") wherever you need it.
21//
22// jassertfalse fires in Debug if a string ID is not found — helps catch typos.
23// =====================================================================
24class LanguageManager
25{
26public:
27 enum class Language { English, Japanese };
28
29 static LanguageManager& getInstance() noexcept
30 {
31 static LanguageManager instance;
32 return instance;
33 }
34
35 // Call once during app initialise to restore the saved language.
36 void init (juce::PropertiesFile* prefs)
37 {
38 if (prefs != nullptr)
39 current_ = (prefs->getValue ("language", "en") == "ja")
40 ? Language::Japanese : Language::English;
41 }
42
43 // Change language and optionally persist the choice.
44 void setLanguage (Language lang, juce::PropertiesFile* prefs = nullptr)
45 {
46 current_ = lang;
47 if (prefs != nullptr)
48 prefs->setValue ("language",
49 lang == Language::Japanese ? "ja" : "en");
50 }
51
52 Language getLanguage() const noexcept { return current_; }
53 bool isJapanese() const noexcept { return current_ == Language::Japanese; }
54
