17// PC keyboard → MIDI note (standard convention: MIDI 60 = C4; Z = C5 = MIDI 72)
18inline int pcKeyToNote (int kc)
19{
20 switch (kc)
21 {
22 case 'z': case 'Z': return 72;
23 case 's': case 'S': return 73;
24 case 'x': case 'X': return 74;
25 case 'd': case 'D': return 75;
26 case 'c': case 'C': return 76;
27 case 'v': case 'V': return 77;
28 case 'g': case 'G': return 78;
29 case 'b': case 'B': return 79;
30 case 'h': case 'H': return 80;
31 case 'n': case 'N': return 81;
32 case 'j': case 'J': return 82;
33 case 'm': case 'M': return 83;
34 case ',': return 84;
35 default: return -1;
36 }
37}

207 // Metronome icon — trapezoid body + tick mark + diagonal pendulum arm + bob
208 inline void metronome (Graphics& g, Rectangle<float> a)
209 {
210 g.setColour (Colours::white);
211
212 const float cx = a.getCentreX();
213 const float top = a.getY() + a.getHeight() * 0.05f;
214 const float bottom = a.getBottom() - a.getHeight() * 0.05f;
215 const float hh = (bottom - top) * 0.5f;
216 const float cy = top + hh;
217 const float hwB = a.getWidth() * 0.38f; // half-width at base
218 const float hwT = a.getWidth() * 0.22f; // half-width at top
219
220 // Trapezoid body (outline)
221 juce::Path body;
222 body.startNewSubPath (cx - hwB, bottom);
223 body.lineTo (cx + hwB, bottom);
224 body.lineTo (cx + hwT, top);
225 body.lineTo (cx - hwT, top);
226 body.closeSubPath();
227 g.strokePath (body, juce::PathStrokeType (1.4f));
228
229 // Small tick mark at top centre
230 g.drawLine (cx, top, cx, top + a.getHeight() * 0.12f, 1.4f);
231
232 // Pendulum arm — pivot at base-centre, bob tilted right
233 const float pivotY = bottom - a.getHeight() * 0.10f;
234 const float bobX = cx + hwT * 0.80f;
235 const float bobY = cy - hh * 0.20f;
236 g.drawLine (cx, pivotY, bobX, bobY, 1.6f);
237
238 // Bob (small filled circle at end of arm)
239 g.fillEllipse (bobX - 2.2f, bobY - 2.2f, 4.4f, 4.4f);

242 // Star icon — 5-point star, filled (★) or outline (☆)
243 inline void star (Graphics& g, Rectangle<float> a, bool filled)
244 {
245 const float cx = a.getCentreX();
246 const float cy = a.getCentreY() + a.getHeight() * 0.04f; // slight downward shift for visual balance
247 const float r1 = a.getWidth() * 0.44f; // outer radius
248 const float r2 = r1 * 0.42f; // inner radius
249 const int pts = 5;
250
251 Path star;
252 for (int i = 0; i < pts * 2; ++i)
253 {
254 float angle = (float(i) / float(pts * 2)) * MathConstants<float>::twoPi
255 - MathConstants<float>::halfPi;
256 float r = (i % 2 == 0) ? r1 : r2;
257 float x = cx + r * std::cos (angle);
258 float y = cy + r * std::sin (angle);
259 if (i == 0) star.startNewSubPath (x, y);
260 else star.lineTo (x, y);
261 }
262 star.closeSubPath();
263
264 if (filled)
265 {
266 g.setColour (Colour (0xffffcc00));
267 g.fillPath (star);
268 }
269 else
270 {
271 g.setColour (Colours::white.withAlpha (0.55f));
272 g.strokePath (star, PathStrokeType (1.2f));
273 }
274 }

277 inline void visualizer (Graphics& g, Rectangle<float> a)
278 {
279 g.setColour (Colours::white);
280 const float cx = a.getCentreX();
281 const float cy = a.getCentreY();
282 const float r = a.getWidth() * 0.30f;
283
284 // Outer ring
285 g.drawEllipse (cx - r, cy - r, r * 2.f, r * 2.f, 1.0f);
286
287 // Radial bars at 8 angles, varying heights
288 const float barHeights[] = { 0.55f, 0.80f, 0.45f, 1.00f,
289 0.65f, 0.90f, 0.50f, 0.70f };
290 const int numBars = 8;
291 const float barMaxLen = a.getWidth() * 0.22f;
292 for (int i = 0; i < numBars; ++i)
293 {
294 float angle = (float)i / (float)numBars * juce::MathConstants<float>::twoPi;
295 float bLen = barMaxLen * barHeights[i];
296 float x1 = cx + r * std::cos (angle);
297 float y1 = cy + r * std::sin (angle);
298 float x2 = cx + (r + bLen) * std::cos (angle);
299 float y2 = cy + (r + bLen) * std::sin (angle);
300 g.drawLine (x1, y1, x2, y2, 1.5f);
301 }

304 // VU meter icon — semi-circular scale arc + needle + pivot dot
305 inline void vuMeter (Graphics& g, Rectangle<float> a)
306 {
307 g.setColour (Colours::white);
308
309 const float cx = a.getCentreX();
310 const float py = a.getBottom() - a.getHeight() * 0.15f; // pivot y

312 const float arcFrom = juce::MathConstants<float>::pi * 1.15f; // ~207°

318 g.strokePath (arc, juce::PathStrokeType (1.5f));
319
320 // Needle at ~70% deflection (pointing upper-right)
321 const float angle = arcFrom + (arcTo - arcFrom) * 0.70f;
322 const float needleLen = r * 0.88f;
323 const float nx = cx + needleLen * std::cos (angle);
324 const float ny = py + needleLen * std::sin (angle);
325 g.drawLine (cx, py, nx, ny, 1.5f);
326
327 // Pivot dot
328 g.fillEllipse (cx - 2.f, py - 2.f, 4.f, 4.f);

332// =====================================================================
333// LvhLookAndFeel
334//
335// Application-wide LookAndFeel that replaces the JUCE default fonts
336// with Japanese-capable alternatives on Windows 10/11:
337//
338// Default sans-serif → "Yu Gothic UI" (system UI font, W10/11)
339// Default monospace → "MS Gothic" (Japanese-capable fixed-width)
340//
341// Applied globally in LvhProApplication::initialise() so that ALL
342// components — Labels, Buttons, AlertWindows, custom paint() calls
343// using Font(size, style) — render Japanese (and other CJK) text
344// without relying on font-fallback mechanisms that JUCE may not invoke.
345// =====================================================================
346class LvhLookAndFeel : public LookAndFeel_V4
347{
348public:
349 Typeface::Ptr getTypefaceForFont (const Font& f) override
350 {
351 const auto& name = f.getTypefaceName();
352
353 if (name == Font::getDefaultSansSerifFontName())
354 return LookAndFeel_V4::getTypefaceForFont (
355 Font ("Yu Gothic UI", f.getHeight(), f.getStyleFlags()));
356
357 if (name == Font::getDefaultMonospacedFontName())
358 return LookAndFeel_V4::getTypefaceForFont (
359 Font ("MS Gothic", f.getHeight(), f.getStyleFlags()));
360
361 return LookAndFeel_V4::getTypefaceForFont (f);
362 }
363};
