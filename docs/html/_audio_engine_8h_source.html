17// =========================================================================
18// GainAndMeterProcessor
19// Inserted as a graph node just before the AudioOutput node.
20// Applies master output gain and captures peak L/R levels — all on the
21// audio thread using only atomic operations (no locks needed).
22// =========================================================================
23class GainAndMeterProcessor : public AudioProcessor
24{
25public:
26 GainAndMeterProcessor()
27 : AudioProcessor (BusesProperties()
28 .withInput ("Input", AudioChannelSet::stereo(), true)
29 .withOutput ("Output", AudioChannelSet::stereo(), true))
30 {
31 gain.store (1.f);
32 peaks[0].store (0.f);
33 peaks[1].store (0.f);
34 rms[0].store (0.f);
35 rms[1].store (0.f);
36 }
37
38 void prepareToPlay (double sampleRate, int samplesPerBlock) override
39 {
40 peaks[0].store (0.f);
41 peaks[1].store (0.f);
42 rms[0].store (0.f);
43 rms[1].store (0.f);
44 rmsSmoothed[0] = 0.f;
45 rmsSmoothed[1] = 0.f;
46
47 // IIR smoothing coefficient — 50ms time constant.
48 // Physics engine handles VU ballistics, so we just need a stable short-window RMS.
49 const double tau = 0.05;
50 rmsAlpha = (float)(1.0 - std::exp (-(double)samplesPerBlock / (tau * sampleRate)));
51 }
52 void releaseResources() override {}
53
54 void processBlock (AudioBuffer<float>& buffer, MidiBuffer&) override
55 {
56 float g = gain.load (std::memory_order_relaxed);
57 const int numSamples = buffer.getNumSamples();
58 for (int ch = 0; ch < jmin (2, buffer.getNumChannels()); ++ch)
59 {
60 buffer.applyGain (ch, 0, numSamples, g);
61
62 // Peak
63 float p = buffer.getMagnitude (ch, 0, numSamples);
64 float cur = peaks[ch].load (std::memory_order_relaxed);
65 if (p > cur) peaks[ch].store (p, std::memory_order_relaxed);
66
67 // RMS — block-level then IIR-smoothed
68 const float* data = buffer.getReadPointer (ch);
69 float sumSq = 0.f;
70 for (int i = 0; i < numSamples; ++i)
71 sumSq += data[i] * data[i];
72 float blockRms = std::sqrt (sumSq / (float)numSamples);
73 rmsSmoothed[ch] += rmsAlpha * (blockRms - rmsSmoothed[ch]);
74 rms[ch].store (rmsSmoothed[ch], std::memory_order_relaxed);
75 }
76
77 // ── FFT + Waveform accumulation ─────────────────────────────────
78 // Mix L+R to mono and accumulate into fftAccum_.
79 const float* L = buffer.getReadPointer (0);
80 const float* R = buffer.getNumChannels() > 1 ? buffer.getReadPointer (1) : L;
81 for (int i = 0; i < numSamples; ++i)
82 {
83 fftAccum_[fftPos_] = (L[i] + R[i]) * 0.5f;
84 ++fftPos_;
85 if (fftPos_ >= fftSize_)
86 {
87 // Apply Hann window and run FFT (in-place, frequency-only).
88 std::copy_n (fftAccum_.begin(), fftSize_, fftWork_.begin());
89 window_.multiplyWithWindowingTable (fftWork_.data(), fftSize_);
90 // Zero imaginary part before FFT
91 std::fill (fftWork_.begin() + fftSize_, fftWork_.end(), 0.f);
92 fft_.performFrequencyOnlyForwardTransform (fftWork_.data());
93
94 // Publish under lock (SpinLock — fast, audio-safe)
95 {
96 juce::SpinLock::ScopedLockType lock (fftPubLock_);
97 const float norm = 1.0f / (float)fftSize_;
98 for (int k = 0; k < fftBins_; ++k)
99 fftMag_[k] = fftWork_[k] * norm;
100 std::copy_n (fftAccum_.begin(), fftSize_, waveformPub_.begin());
101 }
102 fftPos_ = 0;
103 }
104 }
105 }
106
107 // ── FFT / Waveform read (message thread or plugin render thread) ─────
108 // Try-lock: if the audio thread is mid-write, returns without blocking.
109 // Returns the number of bins/samples actually written (may be < size).
110 int readFFTMagnitudes (float* buf, int size) const
111 {
112 juce::SpinLock::ScopedTryLockType tryLock (fftPubLock_);
113 if (! tryLock.isLocked()) return 0;
114 const int n = juce::jmin (size, fftBins_);
115 std::copy_n (fftMag_.begin(), n, buf);
116 return n;
117 }
118
119 int readWaveform (float* buf, int size) const
120 {
121 juce::SpinLock::ScopedTryLockType tryLock (fftPubLock_);
122 if (! tryLock.isLocked()) return 0;
123 const int n = juce::jmin (size, fftSize_);
124 std::copy_n (waveformPub_.begin(), n, buf);
125 return n;
126 }
127
128 static constexpr int fftOrder_ = 10;
129 static constexpr int fftSize_ = 1 << fftOrder_; // 1024 samples
130 static constexpr int fftBins_ = fftSize_ / 2; // 512 magnitude bins
131
132 // Call from the message thread only — returns peak since last call and resets.
133 float exchangePeak (int ch)
134 {
135 if (ch < 0 || ch > 1) return 0.f;
136 return peaks[ch].exchange (0.f, std::memory_order_relaxed);
137 }
138
139 // Call from the message thread only — returns latest smoothed RMS.
140 // Uses load (not exchange) so the physics engine always sees the current
141 // IIR-smoothed value regardless of timer/audio-block timing.
142 float exchangeRms (int ch)
143 {
144 if (ch < 0 || ch > 1) return 0.f;
145 return rms[ch].load (std::memory_order_relaxed);
146 }
147
148 void setGain (float g) noexcept { gain.store (g, std::memory_order_relaxed); }
149
150 // AudioProcessor boilerplate
151 const String getName() const override { return "LVH Gain+Meter"; }
152 double getTailLengthSeconds() const override { return 0.0; }
153 bool acceptsMidi() const override { return false; }
154 bool producesMidi() const override { return false; }
155 bool hasEditor() const override { return false; }
156 AudioProcessorEditor* createEditor() override { return nullptr; }
157 int getNumPrograms() override { return 1; }
158 int getCurrentProgram() override { return 0; }
159 void setCurrentProgram (int) override {}
160 const String getProgramName (int) override { return {}; }
161 void changeProgramName (int, const String&) override{}
162 void getStateInformation (MemoryBlock&) override {}
163 void setStateInformation (const void*, int) override{}
164
165private:
166 std::atomic<float> gain;
167 std::atomic<float> peaks[2];
168
169 // RMS (VU meter)
170 std::atomic<float> rms[2];
171 float rmsSmoothed[2] = {}; // IIR state — audio thread only
172 float rmsAlpha = 0.1f;
173
174 // ── FFT pipeline (audio thread writes, message thread reads) ─────────
175 juce::dsp::FFT fft_ { fftOrder_ };
176 juce::dsp::WindowingFunction<float> window_ { (size_t)fftSize_,
177 juce::dsp::WindowingFunction<float>::hann };
178 // Accumulation (audio thread only)
179 std::array<float, fftSize_> fftAccum_ {};
180 std::array<float, fftSize_ * 2> fftWork_ {}; // interleaved for FFT
181 int fftPos_ = 0;
182 // Published data (protected by fftPubLock_)
183 mutable juce::SpinLock fftPubLock_;
184 std::array<float, fftBins_> fftMag_ {};
185 std::array<float, fftSize_> waveformPub_ {};
186
187 JUCE_DECLARE_NON_COPYABLE_WITH_LEAK_DETECTOR (GainAndMeterProcessor)
188};

190// =========================================================================
191// InputGainProcessor
192// Inserted between audioInputNode and the first master FX (or gain/meter).
193// Applies input gain and captures peak L/R levels — audio thread only.
194// =========================================================================
195class InputGainProcessor : public AudioProcessor
196{
197public:
198 InputGainProcessor()
199 : AudioProcessor (BusesProperties()
200 .withInput ("Input", AudioChannelSet::stereo(), true)
201 .withOutput ("Output", AudioChannelSet::stereo(), true))
202 {
203 peaks[0].store (0.f);
204 peaks[1].store (0.f);
205 }
206
207 void prepareToPlay (double, int) override
208 {
209 peaks[0].store (0.f);
210 peaks[1].store (0.f);
211 }
212 void releaseResources() override {}
213
214 void processBlock (AudioBuffer<float>& buffer, MidiBuffer&) override
215 {
216 float g = gain_.load (std::memory_order_relaxed);
217 bool mono = mono_.load (std::memory_order_relaxed);
218 const int numSamples = buffer.getNumSamples();
219 const int numCh = jmin (2, buffer.getNumChannels());
220
221 if (mono && numCh >= 2)
222 {
223 // Apply gain to L, copy to R → centre-panned mono signal
224 buffer.applyGain (0, 0, numSamples, g);
225 buffer.copyFrom (1, 0, buffer, 0, 0, numSamples);
226 float p = buffer.getMagnitude (0, 0, numSamples);
227 float cur = peaks[0].load (std::memory_order_relaxed);
228 if (p > cur) peaks[0].store (p, std::memory_order_relaxed);
229 peaks[1].store (peaks[0].load (std::memory_order_relaxed), std::memory_order_relaxed);
230 }
231 else
232 {
233 for (int ch = 0; ch < numCh; ++ch)
234 {
235 buffer.applyGain (ch, 0, numSamples, g);
236 float p = buffer.getMagnitude (ch, 0, numSamples);
237 float cur = peaks[ch].load (std::memory_order_relaxed);
238 if (p > cur) peaks[ch].store (p, std::memory_order_relaxed);
239 }
240 }
241 }
242
243 float exchangePeak (int ch)
244 {
245 if (ch < 0 || ch > 1) return 0.f;
246 return peaks[ch].exchange (0.f, std::memory_order_relaxed);
247 }
248
249 void setGain (float g) noexcept { gain_.store (g, std::memory_order_relaxed); }
250 void setMonoMode (bool m) noexcept { mono_.store (m, std::memory_order_relaxed); }
251
252 const String getName() const override { return "LVH Input Gain"; }
253 double getTailLengthSeconds() const override { return 0.0; }
254 bool acceptsMidi() const override { return false; }
255 bool producesMidi() const override { return false; }
256 bool hasEditor() const override { return false; }
257 AudioProcessorEditor* createEditor() override { return nullptr; }
258 int getNumPrograms() override { return 1; }
259 int getCurrentProgram() override { return 0; }
260 void setCurrentProgram (int) override {}
261 const String getProgramName (int) override { return {}; }
262 void changeProgramName (int, const String&) override {}
263 void getStateInformation (MemoryBlock&) override {}
264 void setStateInformation (const void*, int) override {}
265
266private:
267 std::atomic<float> gain_ { 1.0f };
268 std::atomic<bool> mono_ { false };
269 std::atomic<float> peaks[2];
270 JUCE_DECLARE_NON_COPYABLE_WITH_LEAK_DETECTOR (InputGainProcessor)
271};

386 void rebuildBridgeGraph (
387 const juce::Array<BridgeInstance*>& instrumentBridges,
388 const juce::Array<BridgeInstance*>& masterEffects,
389 const std::map<BridgeInstance*, juce::Array<BridgeInstance*>>& perChannelFxMap = {},
390 const juce::Array<BridgeInstance*>& inputFxChain = {})
391 {
392 kbProcessor = nullptr;
393 meterGainProcessor = nullptr;
394 metronomeProcessor_ = nullptr;
395 getOrCreateSlot().detach();
396 audioGraph.clear();
397
398 auto outNode = audioGraph.addNode (
399 std::make_unique<AudioProcessorGraph::AudioGraphIOProcessor> (
400 AudioProcessorGraph::AudioGraphIOProcessor::audioOutputNode));
401
402 auto* mgProc = new GainAndMeterProcessor();
403 mgProc->setGain (pendingGain);
404 meterGainProcessor = mgProc;
405 auto mgNode = audioGraph.addNode (std::unique_ptr<GainAndMeterProcessor> (mgProc));
406 auto metroNode = addMetronomeNode();
407
408 // Build signal chain: instruments (parallel mix + per-chan FX) → master effects (serial) → gain/meter
409 AudioProcessorGraph::Node::Ptr lastNode;
410
411 if (! instrumentBridges.isEmpty())
412 {
413 std::vector<MultiSourceBridgeProcessor::BridgeSource> sources;
414 for (auto* b : instrumentBridges)
415 {
416 MultiSourceBridgeProcessor::BridgeSource src { &b->getSharedMemory(), &b->getSyncEvents(), b };
417 auto it = perChannelFxMap.find (b);
418 if (it != perChannelFxMap.end())
419 for (auto* fx : it->second)
420 src.fxChain.push_back ({ &fx->getSharedMemory(), &fx->getSyncEvents(), fx });
421 sources.push_back (std::move (src));
422 }
423 lastNode = audioGraph.addNode (
424 std::make_unique<MultiSourceBridgeProcessor> (std::move (sources)));
425 }
426
427 AudioProcessorGraph::Node::Ptr firstMasterFxNode;
428 for (auto* b : masterEffects)
429 {
430 auto effectNode = audioGraph.addNode (
431 std::make_unique<BridgeEffectProcessor> (
432 b->getSharedMemory(), b->getSyncEvents(), b));
433 if (lastNode != nullptr)
434 for (int ch = 0; ch < 2; ++ch)
435 audioGraph.addConnection ({{lastNode->nodeID, ch}, {effectNode->nodeID, ch}});
436 lastNode = effectNode;
437 if (firstMasterFxNode == nullptr) firstMasterFxNode = effectNode;
438 }
439
440 if (lastNode != nullptr)
441 for (int ch = 0; ch < 2; ++ch)
442 audioGraph.addConnection ({{lastNode->nodeID, ch}, {mgNode->nodeID, ch}});
443
444 // Physical audio input — InputGainProcessor for gain/mute/metering,
445 // followed by optional serial Input FX chain, then summed into master chain.
446 {
447 inputGainProcessor_ = nullptr;
448 auto inNode = audioGraph.addNode (
449 std::make_unique<AudioProcessorGraph::AudioGraphIOProcessor> (
450 AudioProcessorGraph::AudioGraphIOProcessor::audioInputNode));
451 auto* igProc = new InputGainProcessor();
452 igProc->setGain (pendingInputMuted_ ? 0.f : pendingInputGain_);
453 igProc->setMonoMode (pendingInputMono_);
454 inputGainProcessor_ = igProc;
455 auto igNode = audioGraph.addNode (std::unique_ptr<InputGainProcessor> (igProc));
456 for (int ch = 0; ch < 2; ++ch)
457 audioGraph.addConnection ({{inNode->nodeID, ch}, {igNode->nodeID, ch}});
458
459 // Build serial Input FX chain after InputGainProcessor
460 AudioProcessorGraph::Node::Ptr inputLastNode = igNode;
461 for (auto* b : inputFxChain)
462 {
463 auto fxNode = audioGraph.addNode (
464 std::make_unique<BridgeEffectProcessor> (
465 b->getSharedMemory(), b->getSyncEvents(), b));
466 for (int ch = 0; ch < 2; ++ch)
467 audioGraph.addConnection ({{inputLastNode->nodeID, ch}, {fxNode->nodeID, ch}});
468 inputLastNode = fxNode;
469 }
470
471 auto& targetNode = (firstMasterFxNode != nullptr) ? firstMasterFxNode : mgNode;
472 for (int ch = 0; ch < 2; ++ch)
473 audioGraph.addConnection ({{inputLastNode->nodeID, ch}, {targetNode->nodeID, ch}});
474 }
475
476 connectToOutput (mgNode, outNode, metroNode);
477
478 if (lastSampleRate > 0.0 && lastBufferSize > 0)
479 audioGraph.prepareToPlay (lastSampleRate, lastBufferSize);
480 }

511 void loadPlugin (const PluginDescription& desc, double sampleRate, int bufferSize,
512 std::function<void(bool success, const String& nameOrError)> callback)
513 {
514 if (sampleRate > 0.0) lastSampleRate = sampleRate;
515 if (bufferSize > 0) lastBufferSize = bufferSize;
516 getOrCreateSlot().detach();
517
518 formatManager.createPluginInstanceAsync (desc, sampleRate, bufferSize,
519 [this, sampleRate, bufferSize, cb = std::move (callback)]
520 (std::unique_ptr<AudioPluginInstance> instance, const String& error)
521 {
522 if (instance == nullptr)
523 {
524 buildGraphWithSineWave();
525 cb (false, error);
526 return;
527 }
528
529 kbProcessor = nullptr;
530 meterGainProcessor = nullptr;
531 metronomeProcessor_ = nullptr;
532 auto* rawPtr = instance.get();
533 audioGraph.clear();
534
535 auto outNode = audioGraph.addNode (std::make_unique<AudioProcessorGraph::AudioGraphIOProcessor> (AudioProcessorGraph::AudioGraphIOProcessor::audioOutputNode));
536 auto midiNode = audioGraph.addNode (std::make_unique<AudioProcessorGraph::AudioGraphIOProcessor> (AudioProcessorGraph::AudioGraphIOProcessor::midiInputNode));
537
538 auto* kbProc = new MidiInjectionsProcessor (keyboardState);
539 kbProc->setTranspose (pendingTranspose);
540 kbProc->setChannelFilter (pendingChannel);
541 kbProcessor = kbProc;
542 auto kbNode = audioGraph.addNode (std::unique_ptr<MidiInjectionsProcessor> (kbProc));
543 auto plugNode = audioGraph.addNode (std::move (instance));
544
545 auto* mgProc = new GainAndMeterProcessor();
546 mgProc->setGain (pendingGain);
547 meterGainProcessor = mgProc;
548 auto mgNode = audioGraph.addNode (std::unique_ptr<GainAndMeterProcessor> (mgProc));
549 auto metroNode = addMetronomeNode();
550
551 for (int ch = 0; ch < 2; ++ch)
552 audioGraph.addConnection ({{plugNode->nodeID, ch}, {mgNode->nodeID, ch}});
553
554 audioGraph.addConnection ({{midiNode->nodeID, AudioProcessorGraph::midiChannelIndex},
555 {plugNode->nodeID, AudioProcessorGraph::midiChannelIndex}});
556 audioGraph.addConnection ({{kbNode->nodeID, AudioProcessorGraph::midiChannelIndex},
557 {plugNode->nodeID, AudioProcessorGraph::midiChannelIndex}});
558
559 connectToOutput (mgNode, outNode, metroNode);
560
561 audioGraph.prepareToPlay (sampleRate, bufferSize);
562 getOrCreateSlot().attach (rawPtr, plugNode->nodeID);
563 cb (true, rawPtr->getName());
564 });
565 }

699 // Tap tempo: call on every tap; updates BPM from average interval.
700 void tapMetronomeTempo()
701 {
702 const juce::int64 now = juce::Time::currentTimeMillis();
703 tapTimes_.push_back (now);
704 while (tapTimes_.size() > 8
705 || (tapTimes_.size() > 1 && now - tapTimes_.front() > 3000LL))
706 tapTimes_.pop_front();
707 if (tapTimes_.size() < 2) return;
708 double totalMs = static_cast<double> (tapTimes_.back() - tapTimes_.front());
709 double avgMs = totalMs / static_cast<double> (tapTimes_.size() - 1);
710 setMetronomeBpm (juce::jlimit (40.0, 240.0, 60000.0 / avgMs));
711 }
