6// BridgeInstance
7//
8// Encapsulates all Core-side resources for one LVH-Bridge child process:
9// - CoreIpcManager (named pipe server)
10// - SharedMemoryBuffer (shared audio memory)
11// - SyncEvents (named event pair for audio-loop synchronisation)
12//
13// Usage:
14// auto bridge = std::make_unique<BridgeInstance>();
15// bridge->onConnected = [](BridgeInstance* b) { ... };
16// bridge->onDisconnected = [](BridgeInstance* b) { ... };
17// bridge->launch(pluginPath, bridgeExeFile);
18//
19// NOTE: createSyncProcessor() returns a processor whose internals hold
20// *references* into this BridgeInstance's SHM/SyncEvents, so this
21// BridgeInstance must outlive the processor and its containing graph.
22// =====================================================================
