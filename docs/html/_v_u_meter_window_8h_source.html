6// VUMeterComponent
7// Draws a dual-channel (L + R) analogue VU meter face at 60 fps.
8// Geometry: two 190×132 px faces side by side, pivot 30 px below each face.

10// Interaction (Phase D):
11// Bottom kHandleH px strip is the only interactive area (left-drag to move,

13// via Win32 NCHITTEST.

78 static juce::Point<float> toPoint (float physAngle, float radius,
79 float px, float py) noexcept
80 {
81 const float deg = juce::jmap (physAngle, 0.f, 1.f, -kSwingDeg, kSwingDeg);
82 const float rad = juce::degreesToRadians (deg);
83 return { px + radius * std::sin (rad),
84 py - radius * std::cos (rad) };

123 void drawMeter (juce::Graphics& g, int ch,
124 float fx, float fy, const Palette& pal)
125 {
126 juce::Rectangle<float> face (fx, fy, kMeterW, kMeterH);
127 const float pivotX = fx + kMeterW * 0.5f;
128 const float pivotY = fy + kMeterH + kPivotOffY;
129
130 // Face
131 g.setColour (pal.face);
132 g.fillRoundedRectangle (face, 5.f);
133
134 // Red zone arc
135 drawRedArc (g, pivotX, pivotY, pal.redZone);
136
137 // Scale
138 drawScale (g, pivotX, pivotY, pal);
139
140 // Needle
141 const float physAngle = juce::jlimit (0.f, 1.15f,
142 physics_.getNeedleAngle (ch));
143 const auto tip = toPoint (physAngle, kNeedleLen, pivotX, pivotY);
144
145 g.setColour (juce::Colours::black.withAlpha (0.20f));
146 g.drawLine (pivotX + 1.f, pivotY + 1.f, tip.x + 1.f, tip.y + 1.f, 1.8f);
147 g.setColour (pal.needle);
148 g.drawLine (pivotX, pivotY, tip.x, tip.y, 1.8f);
149 g.fillEllipse (pivotX - 3.5f, pivotY - 3.5f, 7.f, 7.f);
150
151 // Glass sheen
152 {
153 juce::ColourGradient sheen (
154 juce::Colours::white.withAlpha (0.22f), fx, fy,
155 juce::Colours::white.withAlpha (0.00f), fx, fy + kMeterH * 0.45f,
156 false);
157 juce::Graphics::ScopedSaveState ss (g);
158 g.reduceClipRegion (face.toNearestInt());
159 g.setGradientFill (sheen);
160 g.fillRoundedRectangle (face, 5.f);
161 }
162
163 // Border
164 g.setColour (pal.frame.brighter (0.25f));
165 g.drawRoundedRectangle (face, 5.f, 1.5f);
166 }

168 void drawRedArc (juce::Graphics& g, float px, float py,
169 juce::Colour col) const
170 {
171 const float zeroA = dbToPhysAngle (0.f);
172 const float r1 = kTickRad - 14.f;
173 const float r2 = kTickRad - 1.f;
174
175 g.setColour (col.withAlpha (0.30f));
176 const int steps = 16;
177 for (int i = 0; i < steps; ++i)
178 {
179 const float t1 = juce::jmap ((float)i, 0.f, (float)steps, zeroA, 1.f);
180 const float t2 = juce::jmap ((float)(i + 1), 0.f, (float)steps, zeroA, 1.f);
181 const auto a = toPoint (t1, r1, px, py);
182 const auto b = toPoint (t1, r2, px, py);
183 const auto c = toPoint (t2, r2, px, py);
184 const auto d = toPoint (t2, r1, px, py);
185 juce::Path seg;
186 seg.addQuadrilateral (a.x, a.y, b.x, b.y, c.x, c.y, d.x, d.y);
187 g.fillPath (seg);
188 }
189 }

191 void drawScale (juce::Graphics& g, float px, float py,
192 const Palette& pal) const
193 {
194 struct Mark { float db; bool major; };
195 static const Mark kMarks[] = {
196 { -40.f, true }, { -30.f, true }, { -20.f, true },
197 { -10.f, true }, { -7.f, false }, { -5.f, false },
198 { -3.f, true }, { -2.f, false }, { -1.f, false },
199 { 0.f, true }, { +1.f, false }, { +2.f, false },
200 { +3.f, true }
201 };
202
203 for (const auto& m : kMarks)
204 {
205 const float pa = dbToPhysAngle (m.db);
206 const float inner = kTickRad - (m.major ? 10.f : 6.f);
207 const auto p1 = toPoint (pa, kTickRad, px, py);
208 const auto p2 = toPoint (pa, inner, px, py);
209
210 const bool isRed = m.db > 0.f;
211 const bool isZero = m.db == 0.f;
212
213 g.setColour (isRed ? pal.redZone
214 : isZero ? pal.zeroDB : pal.text);
215 g.drawLine (p1.x, p1.y, p2.x, p2.y,
216 isZero ? 2.2f : (m.major ? 1.5f : 1.0f));
217
218 if (m.major)
219 {
220 const auto lp = toPoint (pa, kLabelRad, px, py);
221 const juce::String lbl = (m.db > 0.f)
222 ? ("+" + juce::String ((int)m.db))
223 : juce::String ((int)m.db);
224 g.setFont (juce::Font (8.5f));
225 g.setColour (isRed ? pal.redZone : pal.text);
226 g.drawText (lbl, (int)(lp.x - 13), (int)(lp.y - 7),
227 26, 14, juce::Justification::centred);
228 }
229 }
230 }

269// DocumentWindow that hosts VUMeterComponent.
270// Extends the proven DocumentWindow pattern used by MixerWindow /
271// MetronomeWindow in this project, then applies Win32 overlay styles:
272// - Always-on-top (via JUCE setAlwaysOnTop)
273// - 90% opacity (WS_EX_LAYERED / SetLayeredWindowAttributes)
274// - Click-through (WM_NCHITTEST subclass):
275// non-client area (title bar) → handled by original proc as-is
276// client area, bottom kHandleH px → HTCLIENT (drag + right-click)
277// client area, rest → HTTRANSPARENT (click-through)
278// =========================================================================

281public:
282 // Fired when the user clicks the window's close button.
283 // Wire this to update the toolbar toggle button state.
284 std::function<void()> onClose;

286 explicit VUMeterWindow (VUPhysicsEngine& physics)

288 juce::Colour (0xFF2E2416),
289 juce::DocumentWindow::closeButton)
290 {
291 auto* comp = new VUMeterComponent (physics);

293 setUsingNativeTitleBar (true);
